```python
class PersonalLife:
    def __init__(self):
        self.hobbies = ["🎸 Music", "📚 Reading", "♟️ Chess", "⚽ Football"]
        self.interests = ["🚀 Space Tech", "🎨 Design", "🎯 Productivity"]
        self.currently_reading = "Clean Architecture by Robert C. Martin"
        self.favorite_quote = "The only way to do great work is to love what you do"
    
    def weekend_routine(self):
        activities = ["Code side projects", "Learn new tech", "Contribute to OSS"]
        return f"Weekends = {' → '.join(activities)} 🔄"

life = PersonalLife()
print(life.weekend_routine())
# Output: Weekends = Code side projects → Learn new tech → Contribute to OSS 🔄
```
